
CREATE OR REPLACE FUNCTION quillmd_preserve_clinical_row() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE prior jsonb; dept uuid; pat uuid; enc uuid; org uuid; held jsonb; matched uuid[];
BEGIN
 IF current_setting('transaction_isolation') <> 'read committed' THEN
  RAISE EXCEPTION 'Preservation requires read committed isolation';
 END IF;
 PERFORM pg_advisory_xact_lock(716849203517);
 prior=to_jsonb(OLD);
 IF TG_TABLE_NAME='core_encounter' THEN
  IF TG_OP='DELETE' AND OLD.status='finalized' THEN RAISE EXCEPTION 'Finalized document cannot be deleted'; END IF;
  dept=OLD.department_id; pat=OLD.patient_id; enc=OLD.id;
 ELSIF TG_TABLE_NAME='core_patient' THEN dept=OLD.department_id; pat=OLD.id;
 ELSIF TG_TABLE_NAME='core_template' THEN dept=OLD.department_id;
 ELSIF TG_TABLE_NAME='core_department' THEN dept=OLD.id; org=OLD.organization_id;
 ELSIF TG_TABLE_NAME='core_job' THEN
  enc=OLD.encounter_id;
  IF enc IS NULL THEN
   SELECT t.department_id INTO dept FROM core_templateversion v JOIN core_template t ON t.id=v.template_id WHERE v.id=OLD.mapping_version_id;
  END IF;
 ELSIF TG_TABLE_NAME IN ('core_codesuggestion','core_generationbatch') THEN enc=OLD.encounter_id;
 END IF;
 IF enc IS NOT NULL AND dept IS NULL THEN SELECT department_id,patient_id INTO dept,pat FROM core_encounter WHERE id=enc; END IF;
 IF org IS NULL AND dept IS NOT NULL THEN SELECT organization_id INTO org FROM core_department WHERE id=dept; END IF;
 -- Historical identity remains fixed: releasing a hold is not an instruction
 -- to transfer its evidence to another patient/department. Include ancestor
 -- metadata whose movement would detach a held descendant from its old scope.
 SELECT array_agg(h.id ORDER BY h.id) INTO matched FROM core_retentionhold h
 WHERE h.scope='global' OR h.organization_id=org OR h.department_id=dept OR h.patient_id=pat OR h.encounter_id=enc
 OR (TG_TABLE_NAME='core_patient' AND h.encounter_id IN (SELECT id FROM core_encounter WHERE patient_id=pat))
 OR (TG_TABLE_NAME='core_department' AND (
   h.patient_id IN (SELECT id FROM core_patient WHERE department_id=dept)
   OR h.encounter_id IN (SELECT id FROM core_encounter WHERE department_id=dept)))
 OR (TG_TABLE_NAME='core_template' AND (
   h.encounter_id IN (SELECT e.id FROM core_encounter e JOIN core_templateversion v ON v.id=e.template_version_id WHERE v.template_id=OLD.id)
   OR h.patient_id IN (SELECT e.patient_id FROM core_encounter e JOIN core_templateversion v ON v.id=e.template_version_id WHERE v.template_id=OLD.id)));
 IF matched IS NOT NULL AND TG_OP='UPDATE' AND (
    (prior->'department_id') IS DISTINCT FROM (to_jsonb(NEW)->'department_id') OR
    (prior->'organization_id') IS DISTINCT FROM (to_jsonb(NEW)->'organization_id') OR
    (prior->'patient_id') IS DISTINCT FROM (to_jsonb(NEW)->'patient_id') OR
    (prior->'encounter_id') IS DISTINCT FROM (to_jsonb(NEW)->'encounter_id') OR
    (prior->'mapping_version_id') IS DISTINCT FROM (to_jsonb(NEW)->'mapping_version_id')
 ) THEN RAISE EXCEPTION 'Preserved clinical scope cannot move'; END IF;
 SELECT jsonb_agg(h.id ORDER BY h.id) INTO held FROM core_retentionhold h
 WHERE h.id=ANY(matched) AND NOT EXISTS (SELECT 1 FROM core_retentionholdrelease r WHERE r.hold_id=h.id);
 IF held IS NOT NULL THEN
  INSERT INTO core_preservedrecord(id,created_at,source_table,source_id,operation,hold_ids,prior_record)
   VALUES(gen_random_uuid(),clock_timestamp(),TG_TABLE_NAME,OLD.id,TG_OP,held,prior);
 END IF;
 IF TG_OP='DELETE' THEN RETURN OLD; END IF;
 RETURN NEW;
END; $$;

CREATE FUNCTION quillmd_batch_guard() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
 IF TG_OP='DELETE' THEN RAISE EXCEPTION 'Generation plans cannot be deleted'; END IF;
 IF (to_jsonb(NEW)-ARRAY['state','error','finished_at']) IS DISTINCT FROM (to_jsonb(OLD)-ARRAY['state','error','finished_at'])
 THEN RAISE EXCEPTION 'Generation batch identity and plan are immutable'; END IF;
 RETURN NEW;
END; $$;
CREATE TRIGGER generation_batch_immutable BEFORE UPDATE OR DELETE ON core_generationbatch FOR EACH ROW EXECUTE FUNCTION quillmd_batch_guard();
CREATE FUNCTION quillmd_batch_part_guard() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE j core_job; b core_generationbatch; p jsonb;
BEGIN
 SELECT * INTO j FROM core_job WHERE id=NEW.job_id;
 SELECT * INTO b FROM core_generationbatch WHERE id=NEW.batch_id;
 p=b.plan->'parts'->(NEW.ordinal-1);
 IF j.kind IS DISTINCT FROM 'generation_part' OR j.encounter_id IS DISTINCT FROM b.encounter_id
 OR j.user_id IS DISTINCT FROM b.user_id OR j.revision IS DISTINCT FROM b.revision
 OR j.config->'generation_part'->>'batch_id' IS DISTINCT FROM b.id::text
 OR j.config->'generation_part'->>'plan_sha256' IS DISTINCT FROM b.plan_sha256
 OR (j.config->'generation_part'->>'ordinal')::int IS DISTINCT FROM NEW.ordinal
 OR p IS NULL OR NEW.field_ids IS DISTINCT FROM p->'field_ids'
 OR NEW.request_sha256 IS DISTINCT FROM p->>'request_sha256'
 OR NEW.maximum_charge IS DISTINCT FROM (p->>'maximum_charge')::numeric
 THEN RAISE EXCEPTION 'Generation child identity is not bound to its immutable plan'; END IF;
 RETURN NEW;
END; $$;
CREATE TRIGGER generation_batch_part_scope BEFORE INSERT ON core_generationbatchpart FOR EACH ROW EXECUTE FUNCTION quillmd_batch_part_guard();
CREATE FUNCTION quillmd_batch_job_guard() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
 IF OLD.kind='generation_part' AND (NEW.request_key IS DISTINCT FROM OLD.request_key
 OR (OLD.result_path<>'' AND (NEW.result_path IS DISTINCT FROM OLD.result_path OR NEW.result_sha256 IS DISTINCT FROM OLD.result_sha256)))
 THEN RAISE EXCEPTION 'A batch attempt or recorded result cannot be replaced'; END IF;
 RETURN NEW;
END; $$;
CREATE TRIGGER generation_batch_job_identity BEFORE UPDATE ON core_job FOR EACH ROW EXECUTE FUNCTION quillmd_batch_job_guard();
CREATE FUNCTION quillmd_batch_event_guard() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE p core_generationbatchpart; b core_generationbatch; actual numeric; n int;
BEGIN
 SELECT * INTO b FROM core_generationbatch WHERE id=NEW.batch_id;
 IF NEW.part_id IS NOT NULL THEN
 SELECT * INTO p FROM core_generationbatchpart WHERE id=NEW.part_id;
 IF p.batch_id IS DISTINCT FROM NEW.batch_id THEN RAISE EXCEPTION 'Foreign generation child event'; END IF;
 END IF;
 IF NEW.kind='admitted' THEN
 SELECT count(*) INTO n FROM core_generationbatchpart WHERE batch_id=b.id;
 SELECT coalesce(sum(r.amount),0) INTO actual FROM core_reservation r JOIN core_generationbatchpart child ON r.job_id=child.job_id WHERE child.batch_id=b.id AND r.state='held';
 IF n<>jsonb_array_length(b.plan->'parts') OR n<1 OR n>8 OR actual<>b.maximum_charge
 THEN RAISE EXCEPTION 'All batch children and complete holds are required before admission'; END IF;
 END IF;
 IF NEW.kind='send_intent' AND (NEW.part_id IS NULL OR NOT EXISTS(SELECT 1 FROM core_generationbatchevent WHERE batch_id=b.id AND kind='admitted'))
 THEN RAISE EXCEPTION 'A send requires an admitted complete batch'; END IF;
 RETURN NEW;
END; $$;
CREATE TRIGGER generation_batch_event_scope BEFORE INSERT ON core_generationbatchevent FOR EACH ROW EXECUTE FUNCTION quillmd_batch_event_guard();
CREATE UNIQUE INDEX generation_batch_part_event_once ON core_generationbatchevent(part_id,kind) WHERE part_id IS NOT NULL;
CREATE UNIQUE INDEX generation_batch_parent_event_once ON core_generationbatchevent(batch_id,kind) WHERE part_id IS NULL;
CREATE TRIGGER generationbatchpart_append_only BEFORE UPDATE OR DELETE ON core_generationbatchpart FOR EACH ROW EXECUTE FUNCTION quillmd_append_only();
CREATE TRIGGER generationbatchevent_append_only BEFORE UPDATE OR DELETE ON core_generationbatchevent FOR EACH ROW EXECUTE FUNCTION quillmd_append_only();
CREATE TRIGGER generationbatchresponse_append_only BEFORE UPDATE OR DELETE ON core_generationbatchresponse FOR EACH ROW EXECUTE FUNCTION quillmd_append_only();
CREATE TRIGGER generationbatch_no_truncate BEFORE TRUNCATE ON core_generationbatch FOR EACH STATEMENT EXECUTE FUNCTION quillmd_preservation_no_truncate();
CREATE TRIGGER generationbatchpart_no_truncate BEFORE TRUNCATE ON core_generationbatchpart FOR EACH STATEMENT EXECUTE FUNCTION quillmd_preservation_no_truncate();
CREATE TRIGGER generationbatchevent_no_truncate BEFORE TRUNCATE ON core_generationbatchevent FOR EACH STATEMENT EXECUTE FUNCTION quillmd_preservation_no_truncate();
CREATE TRIGGER generationbatchresponse_no_truncate BEFORE TRUNCATE ON core_generationbatchresponse FOR EACH STATEMENT EXECUTE FUNCTION quillmd_preservation_no_truncate();
CREATE TRIGGER generation_batch_preservation BEFORE UPDATE OR DELETE ON core_generationbatch FOR EACH ROW EXECUTE FUNCTION quillmd_preserve_clinical_row();
